// ==UserScript== // @name ARCA Mis Aportes – autocompletar desde ciclo-empleado // @namespace rrhh.ciclo-empleado // @version 1.4 // @description Completa Documento, Apellido y Sexo con los datos que la app de RRHH deja en el portapapeles ("ARCA|dni|apellido|sexo"). No toca el captcha. // @match https://*.afip.gob.ar/* // @match https://*.arca.gob.ar/* // @match https://arca.gob.ar/* // @run-at document-idle // @grant none // ==/UserScript== (function () { 'use strict'; const KEY = 'cicloEmpleadoArca'; let datos = null; // 1) Datos en la URL (#ciclo=...), compatibilidad con la versión anterior const m = location.hash.match(/ciclo=([^&]+)/); if (m) { try { datos = JSON.parse(decodeURIComponent(m[1])); sessionStorage.setItem(KEY, JSON.stringify(datos)); history.replaceState(null, '', location.pathname + location.search); } catch (e) { /* ignorar */ } } if (!datos) { try { datos = JSON.parse(sessionStorage.getItem(KEY) || 'null'); } catch (e) { /* ignorar */ } } function aviso(texto, color) { let b = document.getElementById('ciclo-aviso'); if (!b) { b = document.createElement('div'); b.id = 'ciclo-aviso'; b.style.cssText = 'position:fixed;top:8px;left:50%;transform:translateX(-50%);z-index:99999;padding:8px 14px;border-radius:8px;font:14px Arial,sans-serif;color:#fff;box-shadow:0 2px 8px rgba(0,0,0,.3)'; document.body.appendChild(b); } b.style.background = color || '#0D6B58'; b.textContent = texto; } function setValor(el, valor) { const proto = el.tagName === 'SELECT' ? HTMLSelectElement.prototype : HTMLInputElement.prototype; const setter = Object.getOwnPropertyDescriptor(proto, 'value').set; setter.call(el, valor); el.dispatchEvent(new Event('input', { bubbles: true })); // sin 'change'/'blur': en ASP.NET dispararían un postback y recargarían la página } function campos() { const vis = el => el && el.offsetParent !== null && !el.disabled; // Solo cajas de texto visibles (los radios también tienen "Apellido" en el id y confundían la búsqueda) const txt = [...document.querySelectorAll('input')].filter(el => vis(el) && /^(text|search|number|tel)?$/i.test(el.type || '')); const porNombre = re => txt.find(el => re.test((el.id || '') + ' ' + (el.name || ''))); let doc = porNombre(/ocumento|dni|nrodoc/i); let ape = porNombre(/pellido/i); if (!doc && txt.length >= 2) doc = txt[0]; if (!ape && doc) ape = txt[txt.indexOf(doc) + 1] || null; // el campo que sigue al documento if (ape === doc) ape = null; const sex = [...document.querySelectorAll('select')].filter(vis).find(el => /exo/i.test((el.id || '') + ' ' + (el.name || ''))) || [...document.querySelectorAll('select')].filter(vis)[0] || null; return { doc, ape, sex }; } function completar(d, intento) { intento = intento || 0; // Elegir "Documento, Apellido y Sexo". Si el radio recarga la página, se retoman los datos tras la recarga. const radios = [...document.querySelectorAll('input[type=radio]')]; const rd = radios.find(r => /apellido/i.test((r.labels && r.labels[0] ? r.labels[0].textContent : '') + (r.parentElement ? r.parentElement.textContent : ''))) || radios[1]; if (rd && !rd.checked && intento < 2) { sessionStorage.setItem(KEY, JSON.stringify(d)); rd.click(); setTimeout(() => completar(d, intento + 1), 900); // si no hubo recarga, seguir; si la hubo, se retoma al volver return true; // la página puede recargarse; al volver se completa sin tocar el radio } const { doc, ape, sex } = campos(); if (!doc || !ape) { aviso('No pude identificar los campos de Documento/Apellido. Elegí la opción "Documento, Apellido y Sexo" y tocá el botón de nuevo.', '#B3261E'); return false; } setValor(doc, String(d.dni || '')); setValor(ape, String(d.ap || '')); let faltaSexo = false; if (sex && d.sx) { const op = [...sex.options].find(o => o.text.trim().toUpperCase().startsWith(d.sx)); if (op) setValor(sex, op.value); else faltaSexo = true; } else faltaSexo = true; aviso(faltaSexo ? 'Datos cargados. Elegí el sexo, tildá "No soy un robot" y presioná Continuar.' : 'Datos cargados. Tildá "No soy un robot" y presioná Continuar.'); sessionStorage.removeItem(KEY); sessionStorage.setItem(KEY + 'Hecho', String(Date.now())); return true; } // Los datos viajan por el portapapeles: la app los copia como "ARCA|dni|apellido|sexo" async function desdePortapapeles(silencioso) { try { const t = await navigator.clipboard.readText(); if (silencioso && sessionStorage.getItem(KEY + 'Clip') === t) return false; // ya completado con estos datos: no repetir (evita bucles) const p = t.split('|'); if (p[0] !== 'ARCA' || p.length < 3) { if (!silencioso) aviso('El portapapeles no tiene datos de la app. Tocá "Abrir ARCA" en la app.', '#B3261E'); return false; } sessionStorage.setItem(KEY + 'Clip', t); return completar({ dni: p[1], ap: p[2], sx: p[3] || '' }); } catch (e) { if (!silencioso) aviso('No pude leer el portapapeles (permitilo en el navegador).', '#B3261E'); return false; } } function botonManual() { if (document.getElementById('ciclo-btn')) return; const btn = document.createElement('button'); btn.id = 'ciclo-btn'; btn.textContent = 'Completar desde portapapeles'; btn.style.cssText = 'position:fixed;top:8px;right:8px;z-index:99999;padding:6px 10px;font:13px Arial,sans-serif;cursor:pointer'; btn.onclick = () => desdePortapapeles(false); document.body.appendChild(btn); } // Siempre muestra el botón (así se ve que el script está activo en esta página) botonManual(); if (datos) { completar(datos); return; } // Intento automático; reintenta unos segundos hasta que aparezca el formulario let n = 0; const t = setInterval(async () => { n++; const { doc, ape } = campos(); if (doc && ape) { clearInterval(t); desdePortapapeles(true); } else if (n > 20) clearInterval(t); }, 500); })();
